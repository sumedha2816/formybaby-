    <script>
        let player;
        function onYouTubeIframeAPIReady() {
            player = new YT.Player('yt-player', {
                height: '0',
                width: '0',
                videoId: '0yW7w8F2TVA', // Butterflies - MAX & Ali Gatie
                events: {
                    'onReady': onPlayerReady
                }
            });
        }

        function onPlayerReady(event) {
            // Player is ready, but browser blocks autoplay until click
        }

        const startScreen = document.getElementById('start-screen');
        const mainCard = document.getElementById('mainCard');
        const noBtn = document.getElementById('noBtn');
        const yesBtn = document.getElementById('yesBtn');

        let noScale = 1;
        let yesScale = 1;

        function launchValentine() {
            if (player && player.playVideo) {
                player.playVideo();
            }
            startScreen.style.display = 'none';
            mainCard.style.display = 'block';
            startFloatingEmojis();
        }

        function handleNo() {
            noScale -= 0.2;
            yesScale += 0.3;
            if (noScale < 0.3) {
                noBtn.style.display = 'none';
            } else {
                noBtn.style.transform = `scale(${noScale})`;
                noBtn.style.position = 'absolute';
                noBtn.style.left = Math.random() * 70 + '%';
                noBtn.style.top = Math.random() * 70 + '%';
            }
            yesBtn.style.transform = `scale(${yesScale})`;
            yesBtn.innerText = "SAY YES!! 🥰";
        }

        noBtn.addEventListener('mouseover', handleNo);
        noBtn.addEventListener('click', handleNo);

        yesBtn.addEventListener('click', () => {
            mainCard.innerHTML = `
                <div style="font-size: 4em;">🐶❤️💍</div>
                <h1 style="font-size: 2.5em;">YAYYY! 🎉</h1>
                <p>I knew you'd say yes, <strong>Puppy!</strong></p>
                <div style="background: #fff0f3; padding: 20px; border-radius: 20px; border: 3px dashed #ff4d6d; margin: 15px 0;">
                    <p style="margin: 0; font-size: 0.9em; color: #ff4d6d; font-weight: bold;">URGENT MISSION:</p>
                    <h2 style="margin: 10px 0; letter-spacing: 2px; font-size: 1.8em;">9513915366</h2>
                    <p style="margin: 0; font-weight: bold;">Call your girl now! 📞✨</p>
                </div>
                <p>I love you so much! 💏</p>
            `;
            
            var end = Date.now() + (10 * 1000);
            (function frame() {
              confetti({ particleCount: 5, angle: 60, spread: 55, origin: { x: 0 }, colors: ['#ff4d6d', '#ffffff'] });
              confetti({ particleCount: 5, angle: 120, spread: 55, origin: { x: 1 }, colors: ['#ff4d6d', '#ffffff'] });
              if (Date.now() < end) requestAnimationFrame(frame);
            }());
        });

        function startFloatingEmojis() {
            const emojis = ['🐶', '💖', '✨', '🌸', '🐾', '🦋'];
            setInterval(() => {
                const e = document.createElement('div');
                e.className = 'floating-emoji';
                e.innerHTML = emojis[Math.floor(Math.random() * emojis.length)];
                e.style.left = Math.random() * 100 + 'vw';
                e.style.animationDuration = Math.random() * 2 + 3 + 's';
                document.body.appendChild(e);
                setTimeout(() => e.remove(), 4000);
            }, 350);
        }
